    **B. Define `setEncoderMode(modeId)` repaint function** (port the structure from example lines 565–631, scoped to encoders only):

    ```js
    function setEncoderMode(modeId) {
      const def = encoderModeDefinitions[modeId];
      if (!def) return;
      currentEncoderMode = modeId;

      // Repaint mode-strip button highlight
      document.querySelectorAll('.encoder-mode-btn').forEach(btn => {
        if (btn.dataset.mode === modeId) {
          btn.classList.add('active');
        } else {
          btn.classList.remove('active');
        }
      });

      // Repaint top encoders
      document.querySelectorAll('.top-encoder').forEach(enc => {
        enc.style.backgroundColor = def.topEncoderColor;
      });
      // Repaint device encoders
      document.querySelectorAll('.device-encoder').forEach(enc => {
        enc.style.backgroundColor = def.deviceEncoderColor;
      });

      // Repaint legend description (top-of-page region populated by Plan 01: <div id="legend-desc">)
      const legend = document.getElementById('legend-desc');
      if (legend) {
        legend.innerHTML = `<strong>Encoder Mode:</strong> ${modeId} — ${def.desc}`;
      }
    }
    ```
