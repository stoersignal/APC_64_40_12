    ```js
    // Matrix_Maps.py LED color codes (line 37 of Matrix_Maps.py): 0=off, 1=green, 2=green blink, 3=red, 4=red blink, 5=yellow, 6=yellow blink, 7-127=green
    const LED_COLOR_MAP = {
      0: { name: 'off', css: '#1e293b' },
      1: { name: 'green', css: '#10b981' },
      2: { name: 'green blink', css: '#10b981' },        // animation could be added later; static for now
      3: { name: 'red', css: '#ef4444' },
      4: { name: 'red blink', css: '#ef4444' },
      5: { name: 'yellow', css: '#f59e0b' },
      6: { name: 'yellow blink', css: '#f59e0b' },
    };
    function colorCodeToCss(code) {
      if (code in LED_COLOR_MAP) return LED_COLOR_MAP[code].css;
      if (code >= 7 && code <= 127) return LED_COLOR_MAP[1].css;  // 7-127 = green
      return LED_COLOR_MAP[0].css;
    }
    function colorCodeToName(code) {
      if (code in LED_COLOR_MAP) return LED_COLOR_MAP[code].name;
      if (code >= 7 && code <= 127) return 'green';
      return 'unknown';
    }
    function midiNoteName(n) {
      const names = ['C', 'C#', 'D', 'D#', 'E', 'F', 'F#', 'G', 'G#', 'A', 'A#', 'B'];
      const oct = Math.floor(n / 12) - 2;  // MIDI: C-2 = 0
      return `${names[n % 12]}${oct}`;
    }
    ```
