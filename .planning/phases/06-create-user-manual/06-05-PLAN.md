    **C. Define `toggleModifierOverlay(mode)`:**

    ```js
    function toggleModifierOverlay(mode) {
      modifierOverlays[mode] = !modifierOverlays[mode];

      // Repaint the modifier-overlay-btn highlight
      document.querySelectorAll('.modifier-overlay-btn').forEach(btn => {
        if (btn.dataset.mode === mode) {
          btn.classList.toggle('active', modifierOverlays[mode]);
        }
      });

      // Apply or remove the overlay visual classes
      if (mode === 'ShiftHeld') {
        const def = modifierOverlayDefinitions['ShiftHeld'];
        def.affectedSelectors.forEach(sel => {
          document.querySelectorAll(sel).forEach(el => {
            el.classList.toggle('shift-overlay', modifierOverlays['ShiftHeld']);
          });
        });
      } else if (mode === 'SaveMode') {
        const def = modifierOverlayDefinitions['SaveMode'];
        def.highlightSelectors.forEach(sel => {
          document.querySelectorAll(sel).forEach(el => {
            el.classList.toggle('save-highlight', modifierOverlays['SaveMode']);
          });
        });
        def.dimSelectors.forEach(sel => {
          document.querySelectorAll(sel).forEach(el => {
            el.classList.toggle('save-dimmed', modifierOverlays['SaveMode']);
          });
        });
      }

      // Update legend
      const legend = document.getElementById('legend-desc');
      if (legend) {
        const overlays = Object.keys(modifierOverlays).filter(k => modifierOverlays[k]).join(' + ') || 'none';
        const lines = legend.innerHTML.split('<br>');
        // Keep the encoder + matrix lines (first two), append/replace overlay line
        const base = lines.filter(l => !l.includes('Modifier Overlay')).join('<br>');
        legend.innerHTML = base + `<br><strong>Modifier Overlay:</strong> ${overlays}`;
      }
    }
    ```
